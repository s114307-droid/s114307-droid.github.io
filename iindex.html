import pygame import random # 初始化Pygame pygame.init() # 定義遊戲視窗大小 SCREEN_WIDTH, SCREEN_HEIGHT = 600, 400 screen = pygame.display.set_mode((SCREEN_WIDTH, SCREEN_HEIGHT)) pygame.display.set_caption("挖礦和建設遊戲") # 顏色 WHITE = (255, 255, 255) GREEN = (0, 255, 0) BROWN = (139, 69, 19) BLACK = (0, 0, 0) # 玩家設置 player_size = 40 player_x = SCREEN_WIDTH // 2 player_y = SCREEN_HEIGHT // 2 player_speed = 5 # 資源 resources = {"stone": 0, "wood": 0} # 創建時鐘來控制遊戲更新 clock = pygame.time.Clock() # 定義玩家動作 def move_player(keys): global player_x, player_y if keys[pygame.K_LEFT]: # 左移 player_x -= player_speed if keys[pygame.K_RIGHT]: # 右移 player_x += player_speed if keys[pygame.K_UP]: # 上移 player_y -= player_speed if keys[pygame.K_DOWN]: # 下移 player_y += player_speed # 繪製玩家角色 def draw_player(): pygame.draw.rect(screen, GREEN, (player_x, player_y, player_size, player_size)) # 挖礦功能 def mine(): if random.random() < 0.5: # 50%機會獲得資源 resources["stone"] += 1 print("你獲得了一塊石頭！") # 建設功能 def build(): if resources["wood"] >= 5 and resources["stone"] >= 3: resources["wood"] -= 5 resources["stone"] -= 3 print("你建造了一個小房子！") else: print("資源不足，無法建設房子！") # 遊戲主循環 def game(): global player_x, player_y running = True while running: # 檢查事件 for event in pygame.event.get(): if event.type == pygame.QUIT: running = False # 鍵盤控制 keys = pygame.key.get_pressed() move_player(keys) # 挖礦和建設操作 if keys[pygame.K_m]: # 按M進行挖礦 mine() if keys[pygame.K_b]: # 按B進行建設 build() # 填充背景 screen.fill(WHITE) # 顯示玩家 draw_player() # 顯示資源 font = pygame.font.SysFont(None, 36) resource_text = font.render(f"石頭: {resources['stone']} 木材: {resources['wood']}", True, BLACK) screen.blit(resource_text, (10, 10)) # 更新顯示 pygame.display.update() # 設定更新速度 clock.tick(30) # 遊戲結束 pygame.quit() # 啟動遊戲 if __name__ == "__main__": game()
